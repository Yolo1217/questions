import (
        "fmt"

        "go.uber.org/zap"
        "google.golang.org/grpc"
        "google.golang.org/grpc/credentials/insecure"

        notev1 "github.com/BwCloudWeGo/bw-cli/api/gen/note/v1"
        orderv1 "github.com/BwCloudWeGo/bw-cli/api/gen/order/v1"
        userv1 "github.com/BwCloudWeGo/bw-cli/api/gen/user/v1"
        "github.com/BwCloudWeGo/bw-cli/pkg/config"
)

// Clients 聚合 HTTP gateway 使用的所有 gRPC client。
type Clients struct {
        User   userv1.UserServiceClient
        Note   notev1.NoteServiceClient
        Order  orderv1.OrderServiceClient
        Config *config.Config

        conns []*grpc.ClientConn
}

// New 连接配置的 gRPC 目标地址并创建强类型服务 client。
func New(cfg *config.Config, log *zap.Logger) (*Clients, error) {
        userTarget := cfg.ServiceTarget("user")
        noteTarget := cfg.ServiceTarget("note")
        orderTarget := cfg.ServiceTarget("order")
        userConn, err := grpc.Dial(userTarget, grpc.WithTransportCredentials(insecure.NewCredentials()))
        if err != nil {
                return nil, fmt.Errorf("dial user service: %w", err)
        }
        noteConn, err := grpc.Dial(noteTarget, grpc.WithTransportCredentials(insecure.NewCredentials()))
        if err != nil {
                userConn.Close()
                return nil, fmt.Errorf("dial note service: %w", err)
        }
        orderConn, err := grpc.Dial(orderTarget, grpc.WithTransportCredentials(insecure.NewCredentials()))
        if err != nil {
                userConn.Close()
                noteConn.Close()
                return nil, fmt.Errorf("dial order service: %w", err)
        }

        log.Info("grpc clients initialized",
                zap.String("user_target", userTarget),
                zap.String("note_target", noteTarget),
                zap.String("order_target", orderTarget),
        )
        return &Clients{
                User:   userv1.NewUserServiceClient(userConn),
                Note:   notev1.NewNoteServiceClient(noteConn),
                Order:  orderv1.NewOrderServiceClient(orderConn),
                Config: cfg,
                conns:  []*grpc.ClientConn{userConn, noteConn, orderConn},
        }, nil
}

// Close 释放 gateway 的所有 gRPC client 连接。
func (c *Clients) Close() {
        for _, conn := range c.conns {
                _ = conn.Close()
        }
}

// MockChannelSecret 是模拟渠道的默认共享密钥，需与订单服务 TRADE_CHANNEL_SECRET 一致。
// 生产环境必须通过环境变量注入，绝不进入前端。
const MockChannelSecret = "shop-mock-channel-secret"

// signMockPayNotify 计算渠道回调签名，与订单服务 SignPayNotify 保持一致：
// hex(HMAC-SHA256(secret, "trade_no|order_id|amount_cents"))。
func signMockPayNotify(secret string, tradeNo string, orderID string, amountCents int64) string {
        if secret == "" {
                secret = MockChannelSecret
        }
        mac := hmac.New(sha256.New, []byte(secret))
        mac.Write([]byte(fmt.Sprintf("%s|%s|%d", tradeNo, orderID, amountCents)))
        return hex.EncodeToString(mac.Sum(nil))
}

// MockPayNotify 处理 POST /api/v1/pay/mock：页面“模拟渠道支付”。
// 网关以服务端密钥签名后调用正式回调链路，因此仍会完整走验签、幂等与异常分支。
func (h *OrderHandler) MockPayNotify(c *gin.Context) {
        var req request.MockPayRequest
        if err := c.ShouldBindJSON(&req); err != nil {
                httpx.Error(c, apperrors.InvalidArgument("invalid_request", err.Error()))
                return
        }
        claims := middleware.ClaimsFromContext(c)
        order, err := h.client.GetTradeOrder(outgoingContext(c), &orderv1.GetTradeOrderRequest{Id: req.OrderID})
        if err != nil {
                httpx.Error(c, apperrors.FromGRPC(err))
                return
        }
        // 买家只能为自己的订单发起支付，防止用他人订单号试探金额
        if claims.Role == "buyer" && order.GetBuyerId() != claims.UserID {
                httpx.Error(c, apperrors.New(apperrors.KindUnauthorized, "permission_denied", "越权拒绝：只能支付本人订单"))
                return
        }
        tradeNo := strings.TrimSpace(req.ChannelTradeNo)
        if tradeNo == "" {
                tradeNo = fmt.Sprintf("CH-MOCK-%s-%d", order.GetId(), time.Now().UnixNano()%1_000_000)
        }
        secret := os.Getenv("TRADE_CHANNEL_SECRET")
        resp, err := h.client.PayNotify(outgoingContext(c), &orderv1.PayNotifyRequest{
                ChannelTradeNo: tradeNo,
                OrderId:        order.GetId(),
                AmountCents:    order.GetAmountCents(),
                Signature:      signMockPayNotify(secret, tradeNo, order.GetId(), order.GetAmountCents()),
        })
        if err != nil {
                httpx.Error(c, apperrors.FromGRPC(err))
                return
        }
        h.log.Info("gateway mock pay notify proxied",
                append(logFields(c, order.GetId()), zap.String("trade_no", tradeNo), zap.String("callback_code", resp.GetCode()))...)
        httpx.OK(c, resp)
}

// registerPages 托管题目01 的可操作页面（go:embed 内嵌，随二进制发布）。
// 页面静态资源为公开资源，业务数据仍由 /api/v1 的 JWT 鉴权保护。
func registerPages(r *gin.Engine) {
        pages := web.FS()
        // 根路径直接回写首页内容：FileFromFS 对 "/" 会触发 index.html 重定向，导致循环。
        indexHTML, err := fs.ReadFile(pages, "index.html")
        if err != nil {
                panic("gateway: embed index.html missing: " + err.Error())
        }
        r.GET("/", func(c *gin.Context) { c.Data(http.StatusOK, "text/html; charset=utf-8", indexHTML) })
        for _, page := range []string{"buyer.html", "warehouse.html", "refund.html", "exception.html", "ops.html"} {
                name := page
                r.GET("/"+name, func(c *gin.Context) { c.FileFromFS(name, http.FS(pages)) })
        }
        assets, err := fs.Sub(pages, "assets")
        if err != nil {
                panic("gateway: embed assets missing: " + err.Error())
        }
        r.StaticFS("/assets", http.FS(assets))
}

// NewTradeOrderID 生成订单号：SO + yyyyMMddHHmmss + 8 位随机十六进制。
// 随机位取 8 位（2^32）以避免同秒批量下单时生日碰撞导致主键冲突（偶发下单失败缺陷修复）。
func NewTradeOrderID() string {
        return "SO" + time.Now().Format("20060102150405") + randomHex(4)
}

// NewExceptionPayID 生成支付异常记录号（同样取 8 位随机十六进制防碰撞）。
func NewExceptionPayID() string {
        return "EX" + time.Now().Format("20060102150405") + randomHex(4)
}

func randomHex(n int) string {
        buf := make([]byte, n)
        if _, err := rand.Read(buf); err != nil {
                return "0000"[:n*2]
        }
        return hex.EncodeToString(buf)

// Order 是 order 业务服务的聚合根。
// 业务明确后，请将 Name 和 Description 替换为真实业务字段。
type Order struct {
        ID          string
        Name        string
        Description string
        CreatedAt   time.Time
        UpdatedAt   time.Time
}

// NewOrder 校验输入，并创建带框架管理身份字段的聚合。
func NewOrder(name string, description string) (*Order, error) {
        name = strings.TrimSpace(name)
        description = strings.TrimSpace(description)
        if name == "" {
                return nil, ErrInvalidOrder
        }
        now := time.Now().UTC()
        return &Order{
                ID:          uuid.NewString(),
                Name:        name,
                Description: description,
                CreatedAt:   now,
                UpdatedAt:   now,
        }, nil
}

// Update 修改可变字段，并把校验保留在业务实体内部。
func (item *Order) Update(name string, description string) error {
        name = strings.TrimSpace(name)
        description = strings.TrimSpace(description)
        if item == nil || item.ID == "" || name == "" {
                return ErrInvalidOrder
        }
        item.Name = name
        item.Description = description
        item.UpdatedAt = time.Now().UTC()
        return nil
}

// TradeRepository 定义交易域所需的持久化行为。
// 全部状态迁移通过 TransitionOrder 的条件更新实现：受影响行数为 0 即迁移失败。
type TradeRepository interface {
        // 商品
        ListProducts(ctx context.Context) ([]*Product, error)
        FindProduct(ctx context.Context, id string) (*Product, error)
        SaveProduct(ctx context.Context, product *Product) error
        HasAnyProduct(ctx context.Context) (bool, error)

        // 订单
        PlaceOrderTx(ctx context.Context, buyerID string, buyerName string, productID string, qty int32, keep time.Duration) (*TradeOrder, error)
        FindTradeOrder(ctx context.Context, id string) (*TradeOrder, error)
        ListTradeOrders(ctx context.Context, buyerID string, status string) ([]*TradeOrder, int64, error)
        // TransitionOrder 条件更新状态：WHERE id=? AND status=from。rows=0 返回 (false, nil)。
        // stockDelta>0 时在同一事务内回补商品可售数量；mutate 在抢占成功后调用，
        // 可修改订单字段并返回随事务一起写入的台账条目。
        TransitionOrder(ctx context.Context, orderID string, from string, to string, stockDelta int32, mutate func(order *TradeOrder) ([]Ledger, error)) (bool, error)
        ListExpiredCreated(ctx context.Context, now time.Time) ([]*TradeOrder, error)
        // MarkRefundFailed 记录渠道退款明确失败：订单保持 REFUNDING，仅置失败标记并写失败流水。
        MarkRefundFailed(ctx context.Context, orderID string) (bool, error)
        // AppendLedger 仅追加台账（用于迟到支付退款：订单状态不变、不回补库存）。
        AppendLedger(ctx context.Context, ledger *Ledger) error

        // 回调记录（入口幂等）
        FindCallback(ctx context.Context, tradeNo string) (*PayCallback, error)
        // ListCallbacks 按渠道交易号/订单号关键词与处理状态查询回调记录。
        ListCallbacks(ctx context.Context, keyword string, status string) ([]*PayCallback, error)
        SaveCallback(ctx context.Context, cb *PayCallback) error // 唯一索引冲突返回 ErrDuplicateTradeNo
        UpdateCallbackStatus(ctx context.Context, tradeNo string, status string) error
        // IncrementCallbackHits 重复通知仅计数，不改变处理状态。
        IncrementCallbackHits(ctx context.Context, tradeNo string) error

        // 支付异常表
        SaveException(ctx context.Context, ex *ExceptionPay) error
        FindException(ctx context.Context, id string) (*ExceptionPay, error)
        // FindExceptionByTradeNo 用渠道交易号反查异常记录（无则返回 nil, nil）。
        FindExceptionByTradeNo(ctx context.Context, tradeNo string) (*ExceptionPay, error)
        ListExceptions(ctx context.Context, onlyPending bool) ([]*ExceptionPay, error)
        CloseException(ctx context.Context, id string, handler string, result string) error

        // 台账与汇总
        ListLedgers(ctx context.Context, orderID string) ([]Ledger, error)
        TradeSummary(ctx context.Context) (*Summary, error)
}

// NewExceptionID 生成异常记录 ID。
func NewExceptionID(seq int64) string {
        return "EX" + time.Now().UTC().Format("20060102150405") + "-" + intToPadded(seq)
}

func intToPadded(v int64) string {
        buf := [6]byte{}
        for i := 5; i >= 0; i-- {
                buf[i] = byte('0' + v%10)
                v /= 10
        }
        return string(buf[:])
}

func mapOrderError(err error) error {
        switch {
        case stderrors.Is(err, entity.ErrInvalidOrder):
                return apperrors.InvalidArgument("invalid_order", "invalid order input")
        case stderrors.Is(err, entity.ErrOrderNotFound):
                return apperrors.NotFound("order_not_found", "order not found")
        default:
                return apperrors.Wrap(apperrors.KindInternal, "order_service_error", "order service error", err)
        }
}

// mapTradeError 将交易域业务错误映射为稳定的应用错误（HTTP 4xx / gRPC 状态码）。
func mapTradeError(err error) error {
        switch {
        case stderrors.Is(err, entity.ErrPermissionDenied):
                return apperrors.New(apperrors.KindUnauthorized, "permission_denied", "越权拒绝：只能操作本人订单")
        case stderrors.Is(err, entity.ErrInsufficientStock):
                return apperrors.Conflict("insufficient_stock", "库存不足：条件扣减失败，不产生订单")
        case stderrors.Is(err, entity.ErrProductOffSale):
                return apperrors.Conflict("product_off_sale", "商品已下架，拒绝下单")
        case stderrors.Is(err, entity.ErrProductNotFound):
                return apperrors.NotFound("product_not_found", "商品不存在")
        case stderrors.Is(err, entity.ErrInvalidQty):
                return apperrors.InvalidArgument("invalid_quantity", "每单数量仅 1~2 件")
        case stderrors.Is(err, entity.ErrTradeOrderNotFound):
                return apperrors.NotFound("order_not_found", "订单不存在")
        case stderrors.Is(err, entity.ErrInvalidTransition):
                return apperrors.Conflict("invalid_status_transition", "当前订单状态不允许该操作")
        case stderrors.Is(err, entity.ErrOrderRefunding):
                return apperrors.Conflict("order_refunding", "订单退款审核中，已暂停发货")
        case stderrors.Is(err, entity.ErrOrderAlreadyShipped):
                return apperrors.Conflict("order_already_shipped", "订单已发货，每单仅允许登记一次")
        case stderrors.Is(err, entity.ErrRefundNotAllowed):
                return apperrors.Conflict("refund_not_allowed", "订单已发货，不受理售前退款")
        case stderrors.Is(err, entity.ErrCarrierRequired):
                return apperrors.InvalidArgument("carrier_required", "承运方与运单号均为必填")
        case stderrors.Is(err, entity.ErrReasonRequired):
                return apperrors.InvalidArgument("reason_required", "必须填写原因")
        case stderrors.Is(err, entity.ErrChannelRefundFailed):
                return apperrors.Conflict("channel_refund_failed", "渠道退款失败：订单保持退款中，待处理记录已保留，可重试")
        case stderrors.Is(err, entity.ErrExceptionNotFound):
                return apperrors.NotFound("exception_not_found", "异常记录不存在")
        case stderrors.Is(err, entity.ErrExceptionClosed):
                return apperrors.Conflict("exception_closed", "异常记录已闭环")
        case stderrors.Is(err, entity.ErrExceptionNoRefund):
                return apperrors.InvalidArgument("exception_no_refund", "无主支付无对应订单，请选择联系渠道核实或标记忽略")
        case stderrors.Is(err, entity.ErrCallbackNotFound):
                return apperrors.NotFound("callback_not_found", "该渠道交易号没有回调记录")
        case stderrors.Is(err, entity.ErrOrderClosed):
                return apperrors.Conflict("order_closed", "订单已关闭（取消/超时），不能发货")
        case stderrors.Is(err, entity.ErrOrderRefunded):
                return apperrors.Conflict("order_refunded", "订单已退款，不能发货")
        default:
                return apperrors.Wrap(apperrors.KindInternal, "trade_service_error", "trade service error", err)
        }
}

// PlaceOrderTx 在同一事务内完成：校验商品 → 条件扣减可售数量 → 创建订单 → 写库存预占流水。
// 条件扣减失败（可售不足）直接返回 ErrInsufficientStock，不产生订单，保证并发下不超卖。
func (r *GormRepository) PlaceOrderTx(ctx context.Context, buyerID string, buyerName string, productID string, qty int32, keep time.Duration) (*entity.TradeOrder, error) {
        var order *entity.TradeOrder
        err := r.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
                var product dbmodel.ProductModel
                if err := tx.Where("id = ?", productID).First(&product).Error; err != nil {
                        if errors.Is(err, gorm.ErrRecordNotFound) {
                                return entity.ErrProductNotFound
                        }
                        return err
                }
                if qty < 1 || qty > 2 {
                        return entity.ErrInvalidQty
                }
                if !product.OnSale {
                        return entity.ErrProductOffSale
                }
                now := time.Now()
                res := tx.Model(&dbmodel.ProductModel{}).
                        Where("id = ? AND available >= ?", productID, qty).
                        Updates(map[string]any{"available": gorm.Expr("available - ?", qty), "updated_at": now})
                if res.Error != nil {
                        return res.Error
                }
                if res.RowsAffected == 0 {
                        return entity.ErrInsufficientStock
                }
                order = &entity.TradeOrder{
                        ID:          entity.NewTradeOrderID(),
                        BuyerID:     buyerID,
                        BuyerName:   buyerName,
                        ProductID:   productID,
                        ProductName: product.Name,
                        Qty:         qty,
                        PriceCents:  product.PriceCents,
                        Status:      entity.OrderStatusCreated,
                        CreatedAt:   now,
                        Deadline:    now.Add(keep),
                        Origin:      entity.OrderOriginSale,
                        UpdatedAt:   now,
                }
                if err := tx.Create(toTradeOrderModel(order)).Error; err != nil {
                        return err
                }
                return tx.Create(&dbmodel.TradeLedgerModel{
                        OrderID:   order.ID,
                        Kind:      entity.LedgerStock,
                        Text:      "下单预占库存 " + itoa(qty) + " 件",
                        Status:    entity.OrderStatusCreated,
                        CreatedAt: now,
                }).Error
        })
        if err != nil {
                return nil, err
        }
        return order, nil
}

// TransitionOrder 条件更新实现唯一状态迁移入口：
// UPDATE ... SET status=to WHERE id=? AND status=from；rows=0 即迁移失败（幂等拒绝）。
// stockDelta>0 时在同一事务内回补可售数量，mutate 返回的台账条目同样随事务写入。
func (r *GormRepository) TransitionOrder(ctx context.Context, orderID string, from string, to string, stockDelta int32, mutate func(order *entity.TradeOrder) ([]entity.Ledger, error)) (bool, error) {
        applied := false
        err := r.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
                now := time.Now()
                res := tx.Model(&dbmodel.TradeOrderModel{}).
                        Where("id = ? AND status = ?", orderID, from).
                        Updates(map[string]any{"status": to, "updated_at": now})
                if res.Error != nil {
                        return res.Error
                }
                if res.RowsAffected == 0 {
                        return nil // 条件不满足：已被并发处理或状态不符
                }
                var record dbmodel.TradeOrderModel
                if err := tx.Where("id = ?", orderID).First(&record).Error; err != nil {
                        return err
                }
                order := toTradeOrderDomain(&record)
                order.Status = to
                order.UpdatedAt = now
                var ledgers []entity.Ledger
                if mutate != nil {
                        var err error
                        ledgers, err = mutate(order)
                        if err != nil {
                                return err
                        }
                }
                updates := map[string]any{
                        "status":        to,
                        "paid_at":       nullableTime(order.PaidAt),
                        "closed_reason": order.ClosedReason,
                        "carrier":       order.Carrier,
                        "tracking_no":   order.TrackingNo,
                        "reject_reason": order.RejectReason,
                        "refund_fail":   order.RefundFail,
                        "origin":        order.Origin,
                        "updated_at":    now,
                }
                if err := tx.Model(&dbmodel.TradeOrderModel{}).Where("id = ?", orderID).Updates(updates).Error; err != nil {
                        return err
                }
                if stockDelta > 0 {
                        if err := tx.Model(&dbmodel.ProductModel{}).
                                Where("id = ?", order.ProductID).
                                Updates(map[string]any{
                                        "available":  gorm.Expr("available + ?", stockDelta),
                                        "updated_at": now,
                                }).Error; err != nil {
                                return err
                        }
                }
                for i := range ledgers {
                        entry := ledgers[i]
                        if entry.Time.IsZero() {
                                entry.Time = now
                        }
                        if entry.OrderID == "" {
                                entry.OrderID = orderID
                        }
                        if err := tx.Create(&dbmodel.TradeLedgerModel{
                                OrderID:     entry.OrderID,
                                Kind:        entry.Kind,
                                Text:        entry.Text,
                                AmountCents: entry.AmountCents,
                                TradeNo:     entry.TradeNo,
                                Status:      entry.Status,
                                CreatedAt:   entry.Time,
                        }).Error; err != nil {
                                return err
                        }
                }
                applied = true
                return nil
        })
        return applied, err
}

// TradeSummary 汇总已收款、已退款、可售总量、异常待处理与各状态订单数。
func (r *GormRepository) TradeSummary(ctx context.Context) (*entity.Summary, error) {
        summary := &entity.Summary{OrdersByStatus: map[string]int32{}}
        var paid, refunded int64
        if err := r.db.WithContext(ctx).Model(&dbmodel.TradeLedgerModel{}).
                Where("kind = ?", entity.LedgerPay).
                Select("COALESCE(SUM(amount_cents),0)").Scan(&paid).Error; err != nil {
                return nil, err
        }
        if err := r.db.WithContext(ctx).Model(&dbmodel.TradeLedgerModel{}).
                Where("kind = ? AND status = ?", entity.LedgerRefund, "success").
                Select("COALESCE(SUM(amount_cents),0)").Scan(&refunded).Error; err != nil {
                return nil, err
        }
        summary.PaidTotalCents = paid
        summary.RefundedTotalCents = refunded

        var available int64
        if err := r.db.WithContext(ctx).Model(&dbmodel.ProductModel{}).
                Select("COALESCE(SUM(available),0)").Scan(&available).Error; err != nil {
                return nil, err
        }
        summary.AvailableTotal = int32(available)

        var reserved int64
        if err := r.db.WithContext(ctx).Model(&dbmodel.TradeOrderModel{}).
                Where("status = ?", entity.OrderStatusCreated).
                Select("COALESCE(SUM(qty),0)").Scan(&reserved).Error; err != nil {
                return nil, err
        }
        summary.ReservedTotal = int32(reserved)

        var pending int64
        if err := r.db.WithContext(ctx).Model(&dbmodel.ExceptionPayModel{}).
                Where("status = ?", entity.ExcStatusPending).Count(&pending).Error; err != nil {
                return nil, err
        }
        summary.ExceptionPending = int32(pending)

        type statusRow struct {
                K string `gorm:"column:k"`
                N int32  `gorm:"column:n"`
        }
        var rows []statusRow
        if err := r.db.WithContext(ctx).Model(&dbmodel.TradeOrderModel{}).
                Select("status as k, count(*) as n").Group("status").Scan(&rows).Error; err != nil {
                return nil, err
        }
        for _, row := range rows {
                summary.OrdersByStatus[row.K] = row.N
        }
        return summary, nil
}

func nullableTime(value time.Time) *time.Time {
        if value.IsZero() {
                return nil

        return &value

func itoa(v int32) string {
        if v == 0 {
                return "0"
        }
        negative := v < 0
        if negative {
                v = -v
        }
        buf := make([]byte, 0, 8)
        for v > 0 {
                buf = append([]byte{byte('0' + v%10)}, buf...)
                v /= 10
        }
        if negative {
                buf = append([]byte{'-'}, buf...)
        }
        return string(buf)
}

func normalizePagination(page int32, pageSize int32) (int, int) {
        if page <= 0 {
                page = 1
        }
        if pageSize <= 0 {
                pageSize = 20
        }
        if pageSize > 100 {
                pageSize = 100
        }
        return int((page - 1) * pageSize), int(pageSize)
}

// DefaultChannelSecret 是模拟支付渠道与商城共享的验签密钥默认值（生产必须由配置注入）。
const DefaultChannelSecret = "shop-mock-channel-secret"

// TradeConfig 是交易域策略配置。
type TradeConfig struct {
        // Keep 是待支付订单保留期（服务端时间判定，默认 10 分钟）。
        Keep time.Duration
        // RefundFailFirst 开启后客服第一次退款会「渠道明确失败」，重试才成功（演示 S3）。
        RefundFailFirst bool
        // ChannelSecret 是支付回调 HMAC 验签密钥，渠道与商城共享。
        ChannelSecret string
}

// SignPayNotify 计算渠道回调签名：HMAC-SHA256(secret, "trade_no|order_id|amount_cents") 十六进制小写。
// 商城按同一算法自行计算比对，因此伪造/篡改金额的请求无法通过验签。
func SignPayNotify(secret string, tradeNo string, orderID string, amountCents int64) string {
        if secret == "" {
                secret = DefaultChannelSecret
        }
        mac := hmac.New(sha256.New, []byte(secret))
        _, _ = fmt.Fprintf(mac, "%s|%s|%d", tradeNo, orderID, amountCents)
        return hex.EncodeToString(mac.Sum(nil))
}

// UpsertProduct 运营维护商品：价格与可售数量可调整，已有订单按下单快照不受影响。
func (s *TradeService) UpsertProduct(ctx context.Context, cmd dto.UpsertProductCommand) (*dto.ProductDTO, error) {
        if strings.TrimSpace(cmd.ID) == "" || strings.TrimSpace(cmd.Name) == "" {
                return nil, entity.ErrInvalidQty
        }
        if cmd.PriceCents <= 0 || cmd.Available < 0 {
                return nil, entity.ErrInvalidQty
        }
        product, err := s.repo.FindProduct(ctx, cmd.ID)
        if err != nil && !errors.Is(err, entity.ErrProductNotFound) {
                return nil, err
        }
        if errors.Is(err, entity.ErrProductNotFound) {
                product = &entity.Product{ID: cmd.ID, Total: cmd.Available}
        }
        product.Name = cmd.Name
        product.PriceCents = cmd.PriceCents
        product.Available = cmd.Available
        product.OnSale = cmd.OnSale
        if product.Total < product.Available {
                product.Total = product.Available
        }
        if err := s.repo.SaveProduct(ctx, product); err != nil {
                return nil, err
        }
        s.log.Info("product upserted", zap.String("product_id", product.ID), zap.Int64("price_cents", product.PriceCents), zap.Int32("available", product.Available))
        return dto.FromProduct(product), nil
}

// AuditRefund 客服退款审核：同意走渠道退款（成功才回补库存），拒绝须写原因并恢复待发货。
// 已退款订单重复审核/渠道重复退款通知按幂等返回成功，不重复打款、不回补库存。
func (s *TradeService) AuditRefund(ctx context.Context, cmd dto.AuditRefundCommand) (*dto.AuditRefundResult, error) {
        order, err := s.repo.FindTradeOrder(ctx, cmd.OrderID)
        if err != nil {
                return nil, err
        }
        if order.Status == entity.OrderStatusRefunded {
                detail, err := s.GetOrder(ctx, order.ID)
                if err != nil {
                        return nil, err
                }
                return &dto.AuditRefundResult{
                        Order:     detail,
                        Duplicate: true,
                        Message:   "退款已成功：本次为重复提交或重复退款通知，按幂等返回成功（不重复打款、不回补库存）",
                }, nil
        }
        if order.Status != entity.OrderStatusRefunding {
                return nil, entity.ErrInvalidTransition
        }
        if !cmd.Approve {
                if strings.TrimSpace(cmd.Reason) == "" {
                        return nil, entity.ErrReasonRequired
                }
                applied, err := s.repo.TransitionOrder(ctx, order.ID, entity.OrderStatusRefunding, entity.OrderStatusPaid, 0,
                        func(item *entity.TradeOrder) ([]entity.Ledger, error) {
                                item.RejectReason = cmd.Reason
                                return []entity.Ledger{{
                                        Kind:   entity.LedgerRefund,
                                        Text:   "客服拒绝退款（" + cmd.Operator + "）：" + cmd.Reason + "（订单恢复待发货）",
                                        Status: "rejected",
                                }}, nil
                        })
                if err != nil {
                        return nil, err
                }
                if !applied {
                        return nil, entity.ErrInvalidTransition
                }
                detail, err := s.GetOrder(ctx, order.ID)
                if err != nil {
                        return nil, err
                }
                s.log.Info("refund rejected", zap.String("order_id", order.ID), zap.String("operator", cmd.Operator))
                return &dto.AuditRefundResult{Order: detail, Message: "已拒绝退款，订单恢复待发货"}, nil
        }

        // 渠道明确失败：保持 REFUNDING，保留待处理记录，等待客服重试。
        if s.cfg.RefundFailFirst && !order.RefundFail {
                applied, err := s.repo.MarkRefundFailed(ctx, order.ID)
                if err != nil {
                        return nil, err
                }
                if !applied {
                        return nil, entity.ErrInvalidTransition
                }
                s.log.Warn("channel refund failed", zap.String("order_id", order.ID))
                return nil, entity.ErrChannelRefundFailed
        }

        // 退款成功：仅正常售前退款回补库存；迟到支付退款不回补（此前已释放，回补即超卖）。
        restock := int32(0)
        if order.Origin == entity.OrderOriginSale {
                restock = order.Qty
        }
        applied, err := s.repo.TransitionOrder(ctx, order.ID, entity.OrderStatusRefunding, entity.OrderStatusRefunded, restock,
                func(item *entity.TradeOrder) ([]entity.Ledger, error) {
                        ledgers := []entity.Ledger{{
                                Kind:        entity.LedgerRefund,
                                Text:        "渠道退款成功，原路退回",
                                AmountCents: item.AmountCents(),
                                TradeNo:     "RF" + item.ID,
                                Status:      "success",
                        }}
                        if item.Origin == entity.OrderOriginSale {
                                ledgers = append(ledgers, entity.Ledger{
                                        Kind:   entity.LedgerStock,
                                        Text:   "售前退款成功，回补可售 " + int32String(item.Qty) + " 件",
                                        Status: entity.OrderStatusRefunded,
                                })
                        }
                        return ledgers, nil
                })
        if err != nil {
                return nil, err
        }
        if !applied {
                return nil, entity.ErrInvalidTransition
        }
        detail, err := s.GetOrder(ctx, order.ID)
        if err != nil {
                return nil, err
        }
        s.log.Info("refund succeeded", zap.String("order_id", order.ID), zap.String("operator", cmd.Operator), zap.Int32("restock", restock))
        message := "渠道退款成功，已回补可售数量"
        if restock == 0 {
                message = "渠道退款成功，迟到支付退款不回补库存"
        }
        return &dto.AuditRefundResult{Order: detail, Message: message}, nil
}

// ShipOrder 仓库为合法待发货订单登记一次发货。
func (s *TradeService) ShipOrder(ctx context.Context, cmd dto.ShipOrderCommand) (*dto.TradeOrderDTO, error) {
        if strings.TrimSpace(cmd.Carrier) == "" || strings.TrimSpace(cmd.TrackingNo) == "" {
                return nil, entity.ErrCarrierRequired
        }
        order, err := s.repo.FindTradeOrder(ctx, cmd.OrderID)
        if err != nil {
                return nil, err
        }
        switch order.Status {
        case entity.OrderStatusRefunding:
                return nil, entity.ErrOrderRefunding
        case entity.OrderStatusShipped:
                return nil, entity.ErrOrderAlreadyShipped
        case entity.OrderStatusClosed:
                return nil, entity.ErrOrderClosed
        case entity.OrderStatusRefunded:
                return nil, entity.ErrOrderRefunded
        case entity.OrderStatusPaid:
        default:
                return nil, entity.ErrInvalidTransition
        }
        applied, err := s.repo.TransitionOrder(ctx, order.ID, entity.OrderStatusPaid, entity.OrderStatusShipped, 0,
                func(item *entity.TradeOrder) ([]entity.Ledger, error) {
                        item.Carrier = cmd.Carrier
                        item.TrackingNo = cmd.TrackingNo
                        return []entity.Ledger{{
                                Kind:   entity.LedgerShip,
                                Text:   "仓库发货登记：" + cmd.Carrier + " 运单 " + cmd.TrackingNo + "（操作人 " + cmd.Operator + "）",

        if !applied {
                return nil, entity.ErrInvalidTransition
        }
        s.log.Info("order shipped", zap.String("order_id", order.ID), zap.String("tracking_no", cmd.TrackingNo))
        return s.GetOrder(ctx, order.ID)
}

// PayNotify 处理渠道支付通知，固定六步 + 幂等三道防线。
func (s *TradeService) PayNotify(ctx context.Context, cmd dto.PayNotifyCommand) (*dto.PayNotifyResult, error) {
        if strings.TrimSpace(cmd.TradeNo) == "" || strings.TrimSpace(cmd.OrderID) == "" {
                return &dto.PayNotifyResult{Success: false, Code: "invalid_argument", Message: "缺少渠道交易号或订单号"}, nil
        }
        // 第 1 步：验签（先于任何落库）——伪造流量不产生回调记录，不会占用渠道交易号，
        // 真实回调重试不会被入口幂等误判为“重复”返回成功而不入账（评分整改：原实现先落库后验签存在该缺陷）。
        // 验签失败同样落异常表留痕（防重放证据），渠道流水号可在异常表反查；同流水号的合法重试可直接通过。
        expected := SignPayNotify(s.cfg.ChannelSecret, cmd.TradeNo, cmd.OrderID, cmd.AmountCents)
        provided := strings.ToLower(strings.TrimSpace(cmd.Signature))
        if subtle.ConstantTimeCompare([]byte(expected), []byte(provided)) != 1 {
                if err := s.recordException(ctx, entity.ExcBadSignature, cmd, nil); err != nil {
                        return nil, err
                }
                s.log.Warn("callback signature verify failed", zap.String("trade_no", cmd.TradeNo))
                return &dto.PayNotifyResult{Success: false, Code: "verify_failed", Message: "验签失败：签名缺失或不匹配，不入账，已落异常表留痕（渠道用原流水号重试可通过）"}, nil
        }

        // 第 2 步：持久化回调记录（入口幂等：channel_trade_no 主键冲突即重复通知；仅验签通过的回调才会写入）
        if err := s.repo.SaveCallback(ctx, &entity.PayCallback{
                TradeNo:     cmd.TradeNo,
                OrderID:     cmd.OrderID,
                AmountCents: cmd.AmountCents,
                Status:      entity.CallbackReceived,
        }); err != nil {
                if errors.Is(err, entity.ErrDuplicateTradeNo) {
                        _ = s.repo.IncrementCallbackHits(ctx, cmd.TradeNo)
                        s.log.Info("duplicate callback rejected", zap.String("trade_no", cmd.TradeNo))
                        return &dto.PayNotifyResult{Success: true, Duplicate: true, Code: "duplicate", Message: "重复通知：已处理过，返回成功（入口幂等，不重复入账）"}, nil
                }
                return nil, err
        }

        // 第 3 步：查订单，不存在落「无主支付」
        order, err := s.repo.FindTradeOrder(ctx, cmd.OrderID)
        if errors.Is(err, entity.ErrTradeOrderNotFound) {
                if err := s.recordException(ctx, entity.ExcNoOwner, cmd, nil); err != nil {
                        return nil, err
                }
                _ = s.repo.UpdateCallbackStatus(ctx, cmd.TradeNo, entity.CallbackFailed)
                return &dto.PayNotifyResult{Success: false, Code: "order_not_found", Message: "无主支付：订单不存在，已落异常表（可用渠道交易号反查），返回失败码"}, nil
        }
        if err != nil {
                return nil, err
        }

        // 第 4 步：金额比对，不一致绝不入账
        if cmd.AmountCents != order.AmountCents() {
                if err := s.recordException(ctx, entity.ExcAmountMismatch, cmd, order); err != nil {
                        return nil, err
                }
                _ = s.repo.UpdateCallbackStatus(ctx, cmd.TradeNo, entity.CallbackFailed)
                return &dto.PayNotifyResult{Success: false, Code: "amount_mismatch", Message: "金额不符：不入账、不发货，已落异常表，返回失败码"}, nil
        }

        // 第 5 步：按订单状态分流
        switch order.Status {
        case entity.OrderStatusCreated:
                applied, err := s.repo.TransitionOrder(ctx, order.ID, entity.OrderStatusCreated, entity.OrderStatusPaid, 0,
                        func(item *entity.TradeOrder) ([]entity.Ledger, error) {
                                now := time.Now()
                                item.PaidAt = now
                                return []entity.Ledger{
                                        {Kind: entity.LedgerPay, Text: "支付成功入账（预占转实扣）", AmountCents: item.AmountCents(), TradeNo: cmd.TradeNo, Status: "success", Time: now},
                                        {Kind: entity.LedgerStock, Text: "预占转实扣 " + int32String(item.Qty) + " 件", Status: entity.OrderStatusPaid, Time: now},
                                }, nil
                        })
                if err != nil {
                        return nil, err
                }
                if !applied {
                        // 缺陷整改（独立测试 D-2）：迁移失败不能一律当"重复"返回成功——
                        // 若订单已被超时关单/退款收敛到终态，本笔真实支付将既无流水也无异常，
                        // 资金悬置且渠道不再重推。终态场景必须落「关单后到账」异常闭环。
                        latest, rerr := s.repo.FindTradeOrder(ctx, order.ID)
                        if rerr != nil {
                                return nil, rerr
                        }
                        if latest.Status == entity.OrderStatusClosed || latest.Status == entity.OrderStatusRefunded {
                                if err := s.recordException(ctx, entity.ExcLatePay, cmd, latest); err != nil {
                                        return nil, err
                                }
                                _ = s.repo.UpdateCallbackStatus(ctx, cmd.TradeNo, entity.CallbackDone)
                                s.log.Warn("late pay during close race", zap.String("order_id", order.ID), zap.String("trade_no", cmd.TradeNo))
                                return &dto.PayNotifyResult{Success: true, Code: "success", Message: "支付确认前订单已关闭：落「关单后到账」异常，由客服核实退款"}, nil
                        }
                        // 结果幂等：并发通知中已被其他请求处理
                        _ = s.repo.UpdateCallbackStatus(ctx, cmd.TradeNo, entity.CallbackDone)
                        return &dto.PayNotifyResult{Success: true, Duplicate: true, Code: "duplicate", Message: "并发通知：订单已由其他请求处理，返回成功（状态仅迁移一次）"}, nil
                }
                _ = s.repo.UpdateCallbackStatus(ctx, cmd.TradeNo, entity.CallbackDone)
                s.log.Info("payment accepted", zap.String("order_id", order.ID), zap.String("trade_no", cmd.TradeNo))
                return &dto.PayNotifyResult{Success: true, Code: "success", Message: "支付成功：订单 " + order.ID + " → 待发货"}, nil

        case entity.OrderStatusPaid, entity.OrderStatusShipped, entity.OrderStatusRefunding:
                if err := s.recordException(ctx, entity.ExcDoubleCharge, cmd, order); err != nil {
                        return nil, err
                }
                _ = s.repo.UpdateCallbackStatus(ctx, cmd.TradeNo, entity.CallbackDone)
                return &dto.PayNotifyResult{Success: true, Code: "success", Message: "订单已收款：判定重复扣款，落异常表，返回成功（不重复记款）"}, nil

        default:
                // CLOSED / REFUNDED：不恢复订单、不重新占用库存，落「关单后到账」由客服核实退款
                if err := s.recordException(ctx, entity.ExcLatePay, cmd, order); err != nil {
                        return nil, err
                }
                _ = s.repo.UpdateCallbackStatus(ctx, cmd.TradeNo, entity.CallbackDone)
                return &dto.PayNotifyResult{Success: true, Code: "success", Message: "订单已关闭/已退款：落异常表「关单后到账」，返回成功（订单状态保持不变）"}, nil
        }
}

// HandleException 客服处置支付异常：原路退款 / 联系渠道核实 / 标记忽略（须留原因）。
// 迟到支付退款：订单保持 CLOSED，仅写退款流水，不回补库存。
func (s *TradeService) HandleException(ctx context.Context, cmd dto.HandleExceptionCommand) (*dto.ExceptionPayDTO, error) {
        exception, err := s.repo.FindException(ctx, cmd.ExceptionID)
        if err != nil {
                return nil, err
        }
        if exception.Status != entity.ExcStatusPending {
                return nil, entity.ErrExceptionClosed
        }
        operator := strings.TrimSpace(cmd.Operator)
        if operator == "" {
                operator = "客服"
        }
        reason := strings.TrimSpace(cmd.Reason)
        switch cmd.Action {
        case "refund":
                if exception.Type == entity.ExcNoOwner || exception.OrderID == "" {
                        return nil, entity.ErrExceptionNoRefund
                }
                if reason == "" {
                        reason = "核实后原路退款"
                }
                order, err := s.repo.FindTradeOrder(ctx, exception.OrderID)
                if err != nil {
                        return nil, err
                }
                if err := s.repo.CloseException(ctx, exception.ID, operator, "核实后原路退款成功（"+order.ID+"）；原因："+reason); err != nil {
                        return nil, err
                }
                // 订单保持原状态：仅记录退款结果，不恢复订单、不动库存
                if err := s.repo.AppendLedger(ctx, &entity.Ledger{
                        OrderID:     order.ID,
                        Kind:        entity.LedgerRefund,
                        Text:        "迟到支付退款：原路退回（不回补库存，订单保持 " + order.Status + "）",
                        AmountCents: exception.ChannelAmountCents,
                        TradeNo:     "RF" + exception.TradeNo,
                        Status:      "success",
                        Time:        time.Now(),
                }); err != nil {
                        return nil, err
                }
                if err := s.appendExceptionLedger(ctx, exception, "客服从异常表发起退款（"+operator+"）："+reason); err != nil {
                        return nil, err
                }
                s.log.Info("late payment refunded", zap.String("exception_id", exception.ID), zap.String("order_id", order.ID))
        case "ignore":
                if reason == "" {
                        return nil, entity.ErrReasonRequired
                }
                if err := s.repo.CloseException(ctx, exception.ID, operator, "标记忽略："+reason); err != nil {
                        return nil, err
                }
                if err := s.appendExceptionLedger(ctx, exception, "客服标记忽略（"+operator+"）："+reason); err != nil {
                        return nil, err
                }
        default: // verify
                if reason == "" {
                        return nil, entity.ErrReasonRequired
                }
                if err := s.repo.CloseException(ctx, exception.ID, operator, "已联系渠道核实："+reason); err != nil {
                        return nil, err
                }
                if err := s.appendExceptionLedger(ctx, exception, "客服联系渠道核实（"+operator+"）："+reason); err != nil {
                        return nil, err
                }
        }
        closed, err := s.repo.FindException(ctx, exception.ID)
        if err != nil {
                return nil, err
        }
        return dto.FromExceptionPay(closed), nil
}

// appendExceptionLedger 把异常处置结果写入订单台账，保证运营按订单能查到异常处理记录。

// SweepExpired 关闭已超保留期的待支付订单并释放预占（服务端时间判定，重复执行不多释放）。
func (s *TradeService) SweepExpired(ctx context.Context) (int, error) {
        expired, err := s.repo.ListExpiredCreated(ctx, time.Now())
        if err != nil {
                return 0, err
        }
        closed := 0
        for _, order := range expired {
                applied, err := s.repo.TransitionOrder(ctx, order.ID, entity.OrderStatusCreated, entity.OrderStatusClosed, order.Qty,
                        func(item *entity.TradeOrder) ([]entity.Ledger, error) {
                                item.ClosedReason = "超时关闭（服务端时间判定）"
                                return []entity.Ledger{{
                                        Kind:   entity.LedgerStock,
                                        Text:   "超时关闭，释放预占 " + int32String(item.Qty) + " 件",
                                        Status: entity.OrderStatusClosed,
                                }}, nil
                        })
                if err != nil {
                        return closed, err
                }
                if applied {
                        closed++
                        s.log.Info("order closed by timeout", zap.String("order_id", order.ID))
                }
        }
        return closed, nil
}

func (s *TradeService) recordException(ctx context.Context, excType string, cmd dto.PayNotifyCommand, order *entity.TradeOrder) error {
        // 缺陷整改（独立测试 D-1）：同一渠道流水号、同类型且仍待处理的异常只保留一条。
        // 验签失败的伪造流量可反复重推（回调记录不落库、入口幂等拦不住），若每次都新增
        // 工单会刷爆客服工作台；此处去重，后续重推仅记日志（首个工单仍可反查与闭环）。
        if existing, err := s.repo.FindExceptionByTradeNo(ctx, cmd.TradeNo); err != nil {
                return err
        } else if existing != nil && existing.Type == excType && existing.Status == entity.ExcStatusPending {
                s.log.Warn("duplicate pending exception suppressed",
                        zap.String("trade_no", cmd.TradeNo), zap.String("type", excType), zap.String("existing_id", existing.ID))
                return nil
        }
        systemAmount := int64(-1)
        orderID := ""
        if order != nil {
                systemAmount = order.AmountCents()
                orderID = order.ID
        }
        exception := &entity.ExceptionPay{
                ID:                 entity.NewExceptionPayID(),
                Type:               excType,
                OrderID:            orderID,
                TradeNo:            cmd.TradeNo,
                ChannelAmountCents: cmd.AmountCents,
                SystemAmountCents:  systemAmount,
                Status:             entity.ExcStatusPending,
        }
        if err := s.repo.SaveException(ctx, exception); err != nil {
                return err
        }
        s.log.Warn("payment exception recorded", zap.String("exception_id", exception.ID), zap.String("type", excType), zap.String("trade_no", cmd.TradeNo))
        return nil
}

func int32String(v int32) string {
        if v == 0 {
                return "0"
        }
        negative := v < 0
        if negative {
                v = -v
        }
        buf := make([]byte, 0, 8)
        for v > 0 {
                buf = append([]byte{byte('0' + v%10)}, buf...)
                v /= 10
        }
        if negative {
                buf = append([]byte{'-'}, buf...)
        }
        return string(buf)
}

// CORSConfig 控制 gateway 跨域策略。
type CORSConfig struct {
        AllowOrigins     []string      `mapstructure:"allow_origins" yaml:"allow_origins"`
        AllowMethods     []string      `mapstructure:"allow_methods" yaml:"allow_methods"`
        AllowHeaders     []string      `mapstructure:"allow_headers" yaml:"allow_headers"`
        ExposeHeaders    []string      `mapstructure:"expose_headers" yaml:"expose_headers"`
        AllowCredentials bool          `mapstructure:"allow_credentials" yaml:"allow_credentials"`
        MaxAge           time.Duration `mapstructure:"max_age" yaml:"max_age"`
}

// DefaultCORSConfig 返回本地开发可用的宽松 CORS 默认配置。
func DefaultCORSConfig() CORSConfig {
        return CORSConfig{
                AllowOrigins:  []string{"*"},
                AllowMethods:  []string{http.MethodGet, http.MethodPost, http.MethodPut, http.MethodPatch, http.MethodDelete, http.MethodOptions},
                AllowHeaders:  []string{"Origin", "Content-Type", "Authorization", HeaderRequestID},
                ExposeHeaders: []string{HeaderRequestID},
                MaxAge:        12 * time.Hour,
        }
}

// CORS 应用配置的跨域策略，并处理预检请求。
func CORS(cfg CORSConfig) gin.HandlerFunc {
        cfg = fillCORSDefaults(cfg)

        return func(c *gin.Context) {
                origin := c.GetHeader("Origin")
                if origin == "" {
                        c.Next()
                        return
                }
                if !originAllowed(origin, cfg.AllowOrigins) {
                        c.AbortWithStatus(http.StatusForbidden)
                        return
                }

                if allowAllOrigins(cfg.AllowOrigins) && !cfg.AllowCredentials {
                        c.Header("Access-Control-Allow-Origin", "*")
                } else {
                        c.Header("Access-Control-Allow-Origin", origin)
                        c.Header("Vary", "Origin")
                }
                c.Header("Access-Control-Allow-Methods", strings.Join(cfg.AllowMethods, ", "))
                c.Header("Access-Control-Allow-Headers", strings.Join(cfg.AllowHeaders, ", "))
                c.Header("Access-Control-Expose-Headers", strings.Join(cfg.ExposeHeaders, ", "))
                c.Header("Access-Control-Max-Age", strconv.Itoa(int(cfg.MaxAge.Seconds())))
                if cfg.AllowCredentials {
                        c.Header("Access-Control-Allow-Credentials", "true")
                }
                if c.Request.Method == http.MethodOptions {
                        c.AbortWithStatus(http.StatusNoContent)
                        return
                }
                c.Next()
        }
}

func fillCORSDefaults(cfg CORSConfig) CORSConfig {
        defaults := DefaultCORSConfig()
        if len(cfg.AllowOrigins) == 0 {
                cfg.AllowOrigins = defaults.AllowOrigins
        }
        if len(cfg.AllowMethods) == 0 {
                cfg.AllowMethods = defaults.AllowMethods
        }
        if len(cfg.AllowHeaders) == 0 {
                cfg.AllowHeaders = defaults.AllowHeaders
        }
        if len(cfg.ExposeHeaders) == 0 {
                cfg.ExposeHeaders = defaults.ExposeHeaders
        }
        if cfg.MaxAge == 0 {
                cfg.MaxAge = defaults.MaxAge
        }
        return cfg
}

func originAllowed(origin string, allowOrigins []string) bool {
        for _, allowed := range allowOrigins {
                if allowed == "*" || strings.EqualFold(allowed, origin) {
                        return true
                }
        }
        return false
}

func allowAllOrigins(allowOrigins []string) bool {
        for _, allowed := range allowOrigins {
                if allowed == "*" {
                        return true
                }
        }
        return false
}

const claimsContextKey = "jwt_claims"

// JWTConfig 控制 token 签名和校验。
type JWTConfig struct {
        Secret        string `mapstructure:"secret" yaml:"secret"`
        Issuer        string `mapstructure:"issuer" yaml:"issuer"`
        ExpireSeconds int64  `mapstructure:"expire_seconds" yaml:"expire_seconds"`
}

// JWTClaims 是签名 token 中保存的业务载荷。
type JWTClaims struct {
        UserID string `json:"user_id"`
        Role   string `json:"role"`
        Name   string `json:"name"`
}

type registeredJWTClaims struct {
        JWTClaims
        jwt.RegisteredClaims
}

// JWT 负责签发 token，并校验 Gin 路由中的 Bearer 认证。
type JWT struct {
        cfg JWTConfig
}

// DefaultJWTConfig 返回不含密钥的 JWT 默认配置；Secret 必须来自配置文件。
func DefaultJWTConfig() JWTConfig {
        return JWTConfig{
                Issuer:        "xiaolanshu",
                ExpireSeconds: 7200,
        }
}

// NewJWT 创建已应用默认值的 JWT 中间件实例。
func NewJWT(cfg JWTConfig) *JWT {
        defaults := DefaultJWTConfig()
        if cfg.Issuer == "" {
                cfg.Issuer = defaults.Issuer
        }
        if cfg.ExpireSeconds <= 0 {
                cfg.ExpireSeconds = defaults.ExpireSeconds
        }
        return &JWT{cfg: cfg}
}

// GenerateToken 使用当前实例配置为给定 claims 签发 JWT。
func (j *JWT) GenerateToken(claims JWTClaims) (string, error) {
        if strings.TrimSpace(j.cfg.Secret) == "" {
                return "", errors.New("jwt secret is required")
        }

        now := time.Now()
        token := jwt.NewWithClaims(jwt.SigningMethodHS256, registeredJWTClaims{
                JWTClaims: claims,
                RegisteredClaims: jwt.RegisteredClaims{
                        Issuer:    j.cfg.Issuer,
                        IssuedAt:  jwt.NewNumericDate(now),
                        ExpiresAt: jwt.NewNumericDate(now.Add(time.Duration(j.cfg.ExpireSeconds) * time.Second)),
                },
        })
        return token.SignedString([]byte(j.cfg.Secret))
}

// Auth 校验授权请求头中的 Bearer 令牌，并把声明写入 Gin 上下文。
func (j *JWT) Auth() gin.HandlerFunc {
        return func(c *gin.Context) {
                if strings.TrimSpace(j.cfg.Secret) == "" {
                        abortUnauthorized(c, "jwt_secret_missing", "jwt secret is not configured")
                        return
                }
                tokenText := bearerToken(c.GetHeader("Authorization"))
                if tokenText == "" {
                        abortUnauthorized(c, "missing_token", "missing bearer token")
                        return
                }

                claims := &registeredJWTClaims{}
                token, err := jwt.ParseWithClaims(tokenText, claims, func(token *jwt.Token) (interface{}, error) {
                        if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
                                return nil, jwt.ErrTokenSignatureInvalid
                        }
                        return []byte(j.cfg.Secret), nil
                }, jwt.WithIssuer(j.cfg.Issuer))
                if err != nil || !token.Valid {
                        abortUnauthorized(c, "invalid_token", "invalid bearer token")
                        return
                }

                c.Set(claimsContextKey, claims.JWTClaims)
                c.Next()
        }
}

// GenerateToken 使用配置的密钥为给定 claims 签发 JWT。
func GenerateToken(cfg JWTConfig, claims JWTClaims) (string, error) {
        return NewJWT(cfg).GenerateToken(claims)
}

// JWTAuth 校验授权请求头中的 Bearer 令牌，并把声明写入 Gin 上下文。
func JWTAuth(cfg JWTConfig) gin.HandlerFunc {
        return NewJWT(cfg).Auth()
}

// ClaimsFromContext 返回 JWTAuth 解析出的 JWT claims。
func ClaimsFromContext(c *gin.Context) JWTClaims {
        value, ok := c.Get(claimsContextKey)
        if !ok {
                return JWTClaims{}
        }
        claims, _ := value.(JWTClaims)
        return claims
}

func bearerToken(header string) string {
        parts := strings.Fields(header)
        if len(parts) != 2 || !strings.EqualFold(parts[0], "Bearer") {
                return ""
        }
        return parts[1]
}

func abortUnauthorized(c *gin.Context, code string, message string) {
        c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{
                "error": gin.H{
                        "code":    code,
                        "message": message,
                },
        })
}

// HeaderRequestID 是用于请求关联的公开 HTTP 头。
const HeaderRequestID = "X-Request-ID"

// RequestID 确保每个请求都有稳定的关联 ID。
func RequestID() gin.HandlerFunc {
        return func(c *gin.Context) {
                requestID := c.GetHeader(HeaderRequestID)
                if requestID == "" {
                        requestID = uuid.NewString()
                }
                c.Set("request_id", requestID)
                c.Writer.Header().Set(HeaderRequestID, requestID)
                c.Next()
        }
}

// RequestLogger 在每个请求完成后记录结构化 HTTP 访问日志。
func RequestLogger(log *zap.Logger) gin.HandlerFunc {
        return func(c *gin.Context) {
                start := time.Now()
                c.Next()

                log.Info("http request completed",
                        zap.String("request_id", requestID(c)),
                        zap.String("trace_id", c.GetHeader("traceparent")),
                        zap.String("method", c.Request.Method),
                        zap.String("path", c.Request.URL.Path),
                        zap.String("route", c.FullPath()),
                        zap.Int("status", c.Writer.Status()),
                        zap.String("client_ip", c.ClientIP()),
                        zap.String("user_agent", c.Request.UserAgent()),
                        zap.Float64("latency_ms", float64(time.Since(start).Microseconds())/1000),
                        zap.Int64("request_bytes", c.Request.ContentLength),
                        zap.Int("response_bytes", c.Writer.Size()),
                        zap.String("error_code", errorCode(c)),
                )
        }
}

// RequireRole 在 JWTAuth 之后校验角色，越权直接 401/403 中止，不透传到下游服务。
// 角色与账号在 JWT 中签发，网关据此做唯一入口鉴权。
func RequireRole(roles ...string) gin.HandlerFunc {
        allowed := make(map[string]struct{}, len(roles))
        for _, role := range roles {
                allowed[role] = struct{}{}
        }
        return func(c *gin.Context) {
                claims := ClaimsFromContext(c)
                if claims.UserID == "" {
                        httpx.Error(c, apperrors.Unauthorized("missing_token", "missing bearer token"))
                        c.Abort()
                        return
                }
                if _, ok := allowed[claims.Role]; !ok {
                        httpx.Error(c, apperrors.New(apperrors.KindUnauthorized, "permission_denied",
                                "越权拒绝：角色 "+claims.Role+" 无权访问该接口"))
                        c.Abort()
                        return
                }
                c.Next()
        }
}
